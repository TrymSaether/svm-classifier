\chapter{Dual Derivation}\label{chap:dual_derivation}

\begin{proposition}{The dual problem of the primal problem}{dual-derivation}
	The dual problem of the primal problem~\eqref{eq:primal-problem} is given by:
	\begin{align}
		\min_{\symbf{\alpha}} & \frac{1}{2} \inner*{\symbf{\alpha}, \mathrm{Y} \mathrm{G} \mathrm{Y} \symbf{\alpha}} -  \inner*{\symbf{1}_M, \symbf{\alpha}}
		\quad
		\text{subject to}
		\quad
		\begin{cases}
			0 \leq \symbf{\alpha} \leq C\symbf{1}_M \\
			\symbf{\alpha}^\top Y = 0
		\end{cases}\tag{DP}
	\end{align}
	where \(G = X X^\top\).
\end{proposition}

\begin{proof}[Proof of Proposition~\ref{dual-derivation}]
	\vspace{-1em}
	\paragraph{Primal problem (P)}
	We begin with the soft-margin SVM primal:
	\[
		\min_{\symbf{w}, b, \symbf{\xi}}
		\tfrac12\|\symbf{w}\|_2^2
		+ C\symbf{1}_M^\top \symbf{\xi}
		\quad\text{subject to}\quad
		\begin{cases}
			\symbf{1}_M - Y\left(X\symbf{w} + b\symbf{1}_M\right)-\symbf{\xi}\le\symbf{0}_M, \\
			-\symbf{\xi}\le\symbf{0}_M.
		\end{cases}
	\]
	Denote these constraints as
	\(
	c_1(\symbf{w}, b, \symbf{\xi}) \le \symbf{0}_M
	\)
	and
	\(
	c_2(\symbf{\xi}) \le \symbf{0}_M.
	\)

	\paragraph{Lagrangian}
	We introduce the Lagrange multipliers
	\(\symbf{\alpha}_1,\symbf{\alpha}_2 \ge \symbf{0}_M.
	\)
	The Lagrangian is
	\[
		\mathcal{L}\left(\symbf{w}, b, \symbf{\xi}, \symbf{\alpha}_1,\symbf{\alpha}_2\right) =\tfrac12\|\symbf{w}\|_2^2 +C\symbf{1}_M^\top \symbf{\xi} +\symbf{\alpha}_1^\top c_1(\symbf{w},b,\symbf{\xi}) +\symbf{\alpha}_2^\top c_2(\symbf{\xi}).
	\]
	Because $\mathcal{L}$ is convex in $(\symbf{w},b,\symbf{\xi})$ and concave in $(\symbf{\alpha}_1,\symbf{\alpha}_2)$, the minimax theorem (see von~Neumann\cite{vonNeumann1928}) implies
	\[
		\min_{\symbf{w},b,\symbf{\xi}}\max_{\symbf{\alpha}_1,\symbf{\alpha}_2} \mathcal{L}(\dots) =\max_{\symbf{\alpha}_1,\symbf{\alpha}_2}\min_{\symbf{w},b,\symbf{\xi}} \mathcal{L}(\dots).
	\]

	Hence solving the primal \(\min_{\symbf{w},b,\symbf{\xi}} p(\symbf{w},b,\symbf{\xi})\) is equivalent to \(\min_{\symbf{w},b,\symbf{\xi}}\max_{\symbf{\alpha}_1,\symbf{\alpha}_2}\mathcal{L}=\max_{\symbf{\alpha}_1,\symbf{\alpha}_2}\min_{\symbf{w},b,\symbf{\xi}}\mathcal{L}.\)

	\paragraph{Dual problem (D)}

	We define

	\[
		g(\symbf{\alpha}_1,\symbf{\alpha}_2) =\min_{(\symbf{w},b,\symbf{\xi})}\mathcal{L}(\dots).
	\]

	Taking partial derivatives of $\mathcal{L}$ w.r.t.\ $\symbf{w}$, $b$, and $\symbf{\xi}$ gives:
	
	\[
		\nabla_{\symbf{w}}\mathcal{L}:
		\symbf{w}-X^\top Y\symbf{\alpha}_1 =\symbf{0},
		\quad
		\nabla_{b}\mathcal{L}:
		-\symbf{\alpha}_1^\top Y =\symbf{0}^\top,
		\quad
		\nabla_{\symbf{\xi}}\mathcal{L}:
		C\symbf{1}_M^\top-\symbf{\alpha}_1^\top-\symbf{\alpha}_2^\top
		=\symbf{0}_M^\top.
	\]
	
	\subparagraph{KKT conditions}
	Setting these gradients to zero gives:
	\begin{align*}
		\nabla_{\symbf{w}} \Lcal   & = \symbf{w} -  (X^\top Y) \symbf{\alpha}_1 = \symbf{0}                                  \\
		\nabla_{b} \Lcal           & = - \symbf{\alpha}_1^\top Y = \symbf{0}^\top                                            \\
		\nabla_{\symbf{\xi}} \Lcal & = C \symbf{1}_M^\top - \symbf{\alpha}_1^\top - \symbf{\alpha}_2^\top = \symbf{0}_M^\top
	\end{align*}
	
	which gives us the KKT conditions:
	
	\begin{align*}
		\symbf{w}                         & = X^\top Y \symbf{\alpha}_1                                 \\
		\symbf{\alpha}_1^\top Y           & = \symbf{0}_M^\top                                          \\
		\symbf{0}_M \leq \symbf{\alpha}_1 & + \underset{=\symbf{0}}{\symbf{\alpha}_2} \leq C\symbf{1}_M
	\end{align*}

	Then we have that:
	\[
	\norm*{\symbf{w}}_2^2 = \norm{X^\top Y \symbf{\alpha}_1}_2^2 = \left(X^\top Y \symbf{\alpha}_1\right)^\top \left(X^\top Y \symbf{\alpha}_1\right) = \symbf{\alpha}_1^\top Y [X X^\top] Y \symbf{\alpha}_1 = \inner*{\symbf{\alpha}_1, \mathrm{Y} \mathrm{G} \mathrm{Y} \symbf{\alpha}_1}
	\]

	Substituting this into the Lagrangian gives:

	\[
		g(\symbf{\alpha}_1,\symbf{\alpha}_2) =\tfrac12\symbf{\alpha}_1^\top YGY \symbf{\alpha}_1 -\symbf{\alpha}_1^\top YGY\symbf{\alpha}_1 +\symbf{\alpha}_1^\top \symbf{1}_M = -\tfrac12\symbf{\alpha}_1^\top YGY \symbf{\alpha}_1 +\symbf{\alpha}_1^\top \symbf{1}_M = -\tfrac12\inner*{\symbf{\alpha}_1, \mathrm{Y} \mathrm{G} \mathrm{Y} \symbf{\alpha}_1} + \inner*{\symbf{1}_M, \symbf{\alpha}_1}
	\]
	
	Flipping signs for a minimization form, the dual problem is
	\[
		\min_{\symbf{\alpha}}
		\tfrac12\symbf{\alpha}^\top YGY \symbf{\alpha}
		-\symbf{1}_M^\top \symbf{\alpha}
		\quad\text{subject to}\quad
		0\le\symbf{\alpha}\le C\symbf{1}_M,
		\symbf{\alpha}^\top Y =0.
	\]
	This completes the derivation of the dual problem~\eqref{eq:dual-problem}.\qed
\end{proof}
